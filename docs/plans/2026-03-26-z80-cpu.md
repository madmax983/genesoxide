```rust
//! Zilog Z80 CPU emulation.
//!
//! The Z80 is an 8-bit CPU with a 16-bit address bus. In the Genesis it runs
//! at ~3.58 MHz (master clock / 15) and drives the YM2612 and PSG sound chips.

mod decode;
pub mod execute;

pub use execute::{Bus, execute_instruction};

use serde::{Deserialize, Serialize};

/// Z80 CPU state.
#[derive(Debug, Clone)]
pub struct Z80 {
    // Main registers
    pub a: u8,
    pub f: u8,
    pub b: u8,
    pub c: u8,
    pub d: u8,
    pub e: u8,
    pub h: u8,
    pub l: u8,
    // Shadow registers
    pub a_prime: u8,
    pub f_prime: u8,
    pub b_prime: u8,
    pub c_prime: u8,
    pub d_prime: u8,
    pub e_prime: u8,
    pub h_prime: u8,
    pub l_prime: u8,
    // Index registers
    pub ix: u16,
    pub iy: u16,
    // Control
    pub sp: u16,
    pub pc: u16,
    pub i: u8,
    pub r: u8,
    pub iff1: bool,
    pub iff2: bool,
    pub im: u8,
    pub halted: bool,
    pub cycles: u64,
    // Internal
    pub ei_pending: bool, // EI delays one instruction
}

// Flag bit positions in the F register
pub const FLAG_C: u8 = 0x01;  // Carry
pub const FLAG_N: u8 = 0x02;  // Add/Subtract
pub const FLAG_PV: u8 = 0x04; // Parity/Overflow
pub const FLAG_H: u8 = 0x10;  // Half-carry
pub const FLAG_Z: u8 = 0x40;  // Zero
pub const FLAG_S: u8 = 0x80;  // Sign
// Undocumented flags (bits 3 and 5 of F, copied from result)
pub const FLAG_Y: u8 = 0x20;  // Bit 5
pub const FLAG_X: u8 = 0x08;  // Bit 3

impl Z80 {
    #[must_use]
    pub fn new() -> Self {
        Self {
            a: 0, f: 0, b: 0, c: 0, d: 0, e: 0, h: 0, l: 0,
            a_prime: 0, f_prime: 0, b_prime: 0, c_prime: 0,
            d_prime: 0, e_prime: 0, h_prime: 0, l_prime: 0,
            ix: 0, iy: 0, sp: 0, pc: 0,
            i: 0, r: 0, iff1: false, iff2: false, im: 0,
            halted: false, cycles: 0, ei_pending: false,
        }
    }

    // Register pair accessors
    #[must_use] pub fn af(&self) -> u16 { u16::from(self.a) << 8 | u16::from(self.f) }
    #[must_use] pub fn bc(&self) -> u16 { u16::from(self.b) << 8 | u16::from(self.c) }
    #[must_use] pub fn de(&self) -> u16 { u16::from(self.d) << 8 | u16::from(self.e) }
    #[must_use] pub fn hl(&self) -> u16 { u16::from(self.h) << 8 | u16::from(self.l) }

    pub fn set_af(&mut self, v: u16) { self.a = (v >> 8) as u8; self.f = v as u8; }
    pub fn set_bc(&mut self, v: u16) { self.b = (v >> 8) as u8; self.c = v as u8; }
    pub fn set_de(&mut self, v: u16) { self.d = (v >> 8) as u8; self.e = v as u8; }
    pub fn set_hl(&mut self, v: u16) { self.h = (v >> 8) as u8; self.l = v as u8; }

    /// Returns true if a flag is set.
    #[must_use]
    pub fn flag(&self, mask: u8) -> bool { self.f & mask != 0 }

    /// Sets or clears a flag.
    pub fn set_flag(&mut self, mask: u8, val: bool) {
        if val { self.f |= mask; } else { self.f &= !mask; }
    }

    /// Snapshot for save states.
    #[must_use]
    pub fn snapshot(&self) -> Z80Snapshot {
        Z80Snapshot {
            a: self.a, f: self.f, b: self.b, c: self.c,
            d: self.d, e: self.e, h: self.h, l: self.l,
            a_prime: self.a_prime, f_prime: self.f_prime,
            b_prime: self.b_prime, c_prime: self.c_prime,
            d_prime: self.d_prime, e_prime: self.e_prime,
            h_prime: self.h_prime, l_prime: self.l_prime,
            ix: self.ix, iy: self.iy,
            sp: self.sp, pc: self.pc,
            i: self.i, r: self.r,
            iff1: self.iff1, iff2: self.iff2, im: self.im,
            halted: self.halted, cycles: self.cycles,
        }
    }
}

impl Default for Z80 {

}
```

```rust
//! Z80 instruction executor.

use super::Z80;

/// Memory and I/O port access for the Z80.
pub trait Bus {
    fn read_byte(&mut self, addr: u16) -> u8;
    fn write_byte(&mut self, addr: u16, val: u8);
    fn read_port(&mut self, port: u16) -> u8;
    fn write_port(&mut self, port: u16, val: u8);
}

/// Executes one instruction from the Z80's program counter.
/// Returns the number of T-states consumed.
pub fn execute_instruction(cpu: &mut Z80, bus: &mut dyn Bus) -> u8 {
    let opcode = bus.read_byte(cpu.pc);
    cpu.pc = cpu.pc.wrapping_add(1);

    // Increment R register (lower 7 bits only)
    cpu.r = (cpu.r & 0x80) | ((cpu.r.wrapping_add(1)) & 0x7F);

    match opcode {
        0x00 => 4, // NOP
        _ => {
            // Unknown opcode — treat as NOP for now
            4
        }
    }
}
```

```rust
//!
//! JSON format: array of test cases, each with initial/final CPU+RAM state.
//! Test files are at tests/z80-tests/v1/ (one .json file per opcode).

use std::path::Path;
use serde::Deserialize;
use genesoxide_core::z80::{self, Z80, Bus};

#[derive(Debug, Deserialize)]
pub struct TestCase {
    pub name: String,
    pub initial: TestState,
    #[serde(rename = "final")]
    pub final_state: TestState,
    // cycles field is present but we skip it for now
}

#[derive(Debug, Deserialize)]
pub struct TestState {
    pub pc: u16,
    pub sp: u16,
    pub a: u8,
    pub b: u8,
    pub c: u8,
    pub d: u8,
    pub e: u8,

    pub h: u8,
    pub l: u8,

    pub r: u8,
    pub ei: u8,
    pub wz: u16,

    pub iy: u16,

    pub af_prime: u16,
    #[serde(rename = "bc_")]
    pub bc_prime: u16,
    #[serde(rename = "de_")]
    pub de_prime: u16,
    #[serde(rename = "hl_")]
    pub hl_prime: u16,
    pub im: u8,
    pub p: u8,
    pub q: u8,
    pub iff1: u8,
    pub iff2: u8,
    pub ram: Vec<(u16, u8)>,

/// Simple flat 64KB RAM for testing.
pub struct TestBus {
    pub ram: [u8; 0x10000],
    pub ports: [u8; 256],
}

impl TestBus {
    pub fn new() -> Self {
        Self { ram: [0; 0x10000], ports: [0; 256] }
    }

    pub fn load_state(&mut self, state: &TestState) {
        self.ram.fill(0);
        for &(addr, val) in &state.ram {
            self.ram[addr as usize] = val;
        }
    }
}

impl Bus for TestBus {
    fn read_byte(&mut self, addr: u16) -> u8 { self.ram[addr as usize] }
    fn write_byte(&mut self, addr: u16, val: u8) { self.ram[addr as usize] = val; }
    fn read_port(&mut self, port: u16) -> u8 { self.ports[(port & 0xFF) as usize] }
    fn write_port(&mut self, port: u16, val: u8) { self.ports[(port & 0xFF) as usize] = val; }
}

/// Mismatch between expected and actual state.
#[derive(Debug)]
pub struct Mismatch {
    pub field: String,
    pub expected: u32,
    pub actual: u32,
}

/// Result of a failed test.
#[derive(Debug)]
pub struct TestFailure {
    pub test_name: String,
    pub mismatches: Vec<Mismatch>,
}

fn load_cpu_state(cpu: &mut Z80, state: &TestState) {

    cpu.f = state.f;
    cpu.b = state.b;

    cpu.d = state.d;
    cpu.e = state.e;

    cpu.iy = state.iy;

    cpu.i = state.i;
    cpu.r = state.r;
    cpu.iff1 = state.iff1 != 0;
    cpu.iff2 = state.iff2 != 0;
    cpu.im = state.im;

    cpu.a_prime = (state.af_prime >> 8) as u8;
    cpu.f_prime = state.af_prime as u8;
    cpu.b_prime = (state.bc_prime >> 8) as u8;
    cpu.c_prime = state.bc_prime as u8;
    cpu.d_prime = (state.de_prime >> 8) as u8;
    cpu.e_prime = state.de_prime as u8;
    cpu.h_prime = (state.hl_prime >> 8) as u8;
    cpu.l_prime = state.hl_prime as u8;
    cpu.halted = false;
    cpu.ei_pending = false;
}

fn compare_state(cpu: &Z80, bus: &TestBus, expected: &TestState) -> Vec<Mismatch> {
    let mut mismatches = Vec::new();
    let mut check = |field: &str, expected_val: u32, actual_val: u32| {
        if expected_val != actual_val {
            mismatches.push(Mismatch {
                field: field.to_string(),
                expected: expected_val,
                actual: actual_val,
            });
        }
    };

    check("a", u32::from(expected.a), u32::from(cpu.a));
    check("f", u32::from(expected.f), u32::from(cpu.f));
    check("b", u32::from(expected.b), u32::from(cpu.b));
    check("c", u32::from(expected.c), u32::from(cpu.c));
    check("d", u32::from(expected.d), u32::from(cpu.d));
    check("e", u32::from(expected.e), u32::from(cpu.e));
    check("h", u32::from(expected.h), u32::from(cpu.h));
    check("l", u32::from(expected.l), u32::from(cpu.l));
    check("ix", u32::from(expected.ix), u32::from(cpu.ix));
    check("iy", u32::from(expected.iy), u32::from(cpu.iy));
    check("sp", u32::from(expected.sp), u32::from(cpu.sp));
    check("pc", u32::from(expected.pc), u32::from(cpu.pc));
    check("i", u32::from(expected.i), u32::from(cpu.i));
    check("r", u32::from(expected.r), u32::from(cpu.r));
    check("iff1", u32::from(expected.iff1), u32::from(cpu.iff1));
    check("iff2", u32::from(expected.iff2), u32::from(cpu.iff2));
    check("im", u32::from(expected.im), u32::from(cpu.im));

    // Shadow registers
    let exp_af = expected.af_prime;
    let act_af = cpu.af();
    // Actually shadow = prime registers
    check("a'", u32::from((expected.af_prime >> 8) as u8), u32::from(cpu.a_prime));
    check("f'", u32::from(expected.af_prime as u8), u32::from(cpu.f_prime));
    check("b'", u32::from((expected.bc_prime >> 8) as u8), u32::from(cpu.b_prime));
    check("c'", u32::from(expected.bc_prime as u8), u32::from(cpu.c_prime));
    check("d'", u32::from((expected.de_prime >> 8) as u8), u32::from(cpu.d_prime));
    check("e'", u32::from(expected.de_prime as u8), u32::from(cpu.e_prime));
    check("h'", u32::from((expected.hl_prime >> 8) as u8), u32::from(cpu.h_prime));
    check("l'", u32::from(expected.hl_prime as u8), u32::from(cpu.l_prime));

    // RAM
    for &(addr, val) in &expected.ram {
        let actual = bus.ram[addr as usize];
        if actual != val {
            mismatches.push(Mismatch {
                field: format!("ram[{addr:#06X}]"),
                expected: u32::from(val),
                actual: u32::from(actual),
            });
        }
    }

    mismatches
}

/// Loads test cases from a JSON file.
pub fn load_test_file(path: &Path) -> Vec<TestCase> {
    let data = std::fs::read(path).unwrap_or_else(|e| panic!("Failed to read {}: {e}", path.display()));
    serde_json::from_slice(&data).unwrap_or_else(|e| panic!("Failed to parse {}: {e}", path.display()))
}

/// Runs all test cases from a file. Returns (passed, failed, first N failures).
pub fn run_test_file(path: &Path, max_failures: usize) -> (usize, usize, Vec<TestFailure>) {
    let tests = load_test_file(path);
    let mut passed = 0usize;
    let mut failed = 0usize;
    let mut failures = Vec::new();

    for test in &tests {
        let mut cpu = Z80::new();
        let mut bus = TestBus::new();

        load_cpu_state(&mut cpu, &test.initial);
        bus.load_state(&test.initial);

        z80::execute::execute_instruction(&mut cpu, &mut bus);

        let mismatches = compare_state(&cpu, &bus, &test.final_state);
        if mismatches.is_empty() {
            passed += 1;
        } else {
            failed += 1;
            if failures.len() < max_failures {
                failures.push(TestFailure {
                    test_name: test.name.clone(),
                    mismatches,
                });
            }
        }
    }

    (passed, failed, failures)
}
```

```rust
//! Integration tests for jsmoo Z80 test vectors.

use std::path::{Path, PathBuf};
use genesoxide_test_harness::z80_tests;

fn test_data_dir() -> PathBuf {
    let manifest = Path::new(env!("CARGO_MANIFEST_DIR"));
    manifest.join("../../tests/z80-tests/v1")
}

fn run_opcode_tests(filename: &str) {
    let path = test_data_dir().join(filename);
    if !path.exists() {
        eprintln!("Skipping {filename}: not found at {}", path.display());
        return;
    }

    let (passed, failed, failures) = z80_tests::run_test_file(&path, 10);
    let total = passed + failed;

    if failed > 0 {
        let mut msg = format!("\n{filename}: {failed}/{total} FAILED\n");
        for f in &failures {
            msg.push_str(&format!("\n  Test: {}\n", f.test_name));
            for m in &f.mismatches {
                msg.push_str(&format!(
                    "    {}: expected {:#010X}, got {:#010X}\n",
                    m.field, m.expected, m.actual
                ));
            }
        }
        panic!("{msg}");
    }
    eprintln!("{filename}: {passed}/{total} passed");
}

#[test]
fn nop() {
    run_opcode_tests("00.json");
}

/// Full suite runner — runs every JSON file in the test directory.
/// Use: cargo test -p genesoxide-test-harness --test z80_suite full_suite -- --nocapture --ignored
#[test]
#[ignore]
fn full_suite() {
    let dir = test_data_dir();
    if !dir.exists() {
        eprintln!("Z80 test data not found at {}, skipping", dir.display());
        return;
    }

    let mut entries: Vec<_> = std::fs::read_dir(&dir)
        .unwrap()
        .filter_map(|e| e.ok())
        .filter(|e| e.path().extension().is_some_and(|ext| ext == "json"))
        .collect();
    entries.sort_by_key(|e| e.file_name());

    let mut total_passed = 0usize;
    let mut total_failed = 0usize;
    let mut failed_opcodes = Vec::new();

    for entry in &entries {
        let filename = entry.file_name();
        let filename = filename.to_string_lossy();
        let path = entry.path();

        let (passed, failed, failures) = z80_tests::run_test_file(&path, 3);
        total_passed += passed;
        total_failed += failed;

        if failed > 0 {
            eprintln!("FAIL  {filename}: {failed}/{} failed", passed + failed);
            for f in &failures {
                eprintln!("      {} — {:?}", f.test_name,
                    f.mismatches.iter().map(|m| &m.field).collect::<Vec<_>>());
            }
            failed_opcodes.push(filename.to_string());
        } else {
            eprintln!("PASS  {filename}: {passed}/{passed} passed");
        }
    }

    eprintln!("\n════════════════════════════════════════════");
    eprintln!("Total: {} passed, {} failed ({} opcodes tested)",
        total_passed, total_failed, entries.len());
    if !failed_opcodes.is_empty() {
        eprintln!("Failed: {}", failed_opcodes.join(", "));
    }
    eprintln!("════════════════════════════════════════════");

    assert_eq!(total_failed, 0, "{total_failed} Z80 tests failed");
}
```

## Task 6: Z80 CB Prefix — Bit Operations

Implement the 256 CB-prefixed opcodes: rotates/shifts on all registers + (HL), bit test/set/reset.

**Files:**
- Modify: `crates/genesoxide-core/src/z80/execute.rs`

**Structure:**
The CB prefix is regular — the second byte encodes: operation (bits 7-6), bit number (bits 5-3), register (bits 2-0).

- `0x00-0x07`: RLC r
- `0x08-0x0F`: RRC r
- `0x10-0x17`: RL r
- `0x18-0x1F`: RR r
- `0x20-0x27`: SLA r
- `0x28-0x2F`: SRA r
- `0x30-0x37`: SLL r (undocumented, shifts 1 into bit 0)
- `0x38-0x3F`: SRL r
- `0x40-0x7F`: BIT b,r (test bit, set Z flag)
- `0x80-0xBF`: RES b,r (reset bit)
- `0xC0-0xFF`: SET b,r (set bit)

Register encoding (bits 2-0): B=0, C=1, D=2, E=3, H=4, L=5, (HL)=6, A=7

When register=6 ((HL)), read from memory at address HL, operate, write back (except BIT which is read-only).

All rotate/shift operations set S,Z,PV(parity),H=0,N=0,C(from shifted bit).
BIT sets Z=(bit is zero), H=1, N=0, other flags are complex (undocumented behavior).

Add z80_suite tests for CB prefix files, run suite, commit:
`feat(z80): implement CB prefix bit operations`

## Task 8: Z80 DD/FD Prefixes — IX/IY Indexed

Implement DD (IX) and FD (IY) prefixed opcodes. These mirror many base opcodes but use IX/IY+d instead of HL.

**Files:**
- Modify: `crates/genesoxide-core/src/z80/execute.rs`

**Structure:**
DD/FD prefixes remap the base instruction set:
- HL references become IX+d or IY+d (with a signed displacement byte)
- H references become IXH/IYH (undocumented)
- L references become IXL/IYL (undocumented)
- Some instructions are unchanged (those not involving HL/H/L)

**Key groups:**
- `DD/FD 21`: LD IX/IY,nn
- `DD/FD 22/2A`: LD (nn),IX/IY and LD IX/IY,(nn)
- `DD/FD 23/2B`: INC/DEC IX/IY
- `DD/FD 09/19/29/39`: ADD IX/IY,rr
- `DD/FD 34/35`: INC/DEC (IX/IY+d)
- `DD/FD 36`: LD (IX/IY+d),n
- `DD/FD 46/4E/...`: LD r,(IX/IY+d)
- `DD/FD 70-77`: LD (IX/IY+d),r
- `DD/FD 86/8E/96/9E/A6/AE/B6/BE`: ALU A,(IX/IY+d)
- `DD/FD E1/E5`: POP/PUSH IX/IY
- `DD/FD E9`: JP (IX/IY)
- `DD/FD F9`: LD SP,IX/IY
- `DD/FD E3`: EX (SP),IX/IY

**Implementation approach:** Factor the execute function to accept an "index register override" parameter. When a DD/FD prefix is seen, read the next opcode byte and re-dispatch with IX/IY in place of HL.

Add z80_suite tests for DD and FD prefix files, run suite, commit:
`feat(z80): implement DD/FD prefix IX/IY indexed operations`

## Task 9: Z80 DD CB / FD CB — Indexed Bit Operations

Implement the DD CB and FD CB double-prefixed opcodes. These combine IX/IY indexing with bit operations.

**Files:**
- Modify: `crates/genesoxide-core/src/z80/execute.rs`

**Structure:**
Encoding: `DD CB dd op` where dd = signed displacement, op = CB-style operation.

The operation always targets (IX+d) or (IY+d) in memory. For non-BIT operations, the result is ALSO stored in a register (undocumented behavior): the register is encoded in bits 2-0 of the opcode, unless it's 6 ((IX/IY+d) only, no register copy).

Same operations as CB: RLC, RRC, RL, RR, SLA, SRA, SLL, SRL, BIT, RES, SET.

`feat(z80): implement DD CB/FD CB indexed bit operations`
